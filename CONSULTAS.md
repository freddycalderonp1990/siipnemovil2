SELECT
e.idHdrEvento,
e.idGenGeoSenplades,
e.idGenPersona,
e.usuario,
e.fechaEvento
FROM
hdrEvento e
WHERE e.usuario = 55948
AND e.estadoPolicia = '3'
AND e.idGenPersona = 724115
AND e.ip = 'movil'
